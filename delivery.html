// ត្រង់មុខងារ Confirm Place Order ក្នុង delivery.html function confirmPlaceOrder() { const cart = JSON.parse(localStorage.getItem('cartData')) || []; const customerName = document.getElementById('custName').value; const customerPhone = document.getElementById('custPhone').value; const customerAddress = document.getElementById('custAddress').value; const deliveryFee = parseFloat(document.getElementById('shippingFee').value || 0); if (!customerName || !customerPhone || !customerAddress) { alert('សូមបំពេញព័ត៌មានអតិថិជនឱ្យបានគ្រប់គ្រាន់!'); return; } let itemsTotal = 0; cart.forEach(item => itemsTotal += parseFloat(item.price || 0)); const newOrder = { orderId: "ORD-" + Date.now(), cart: cart, details: { customer: { name: customerName, phone: customerPhone, address: customerAddress }, itemsTotal: itemsTotal, deliveryFee: deliveryFee, grandTotal: itemsTotal + deliveryFee }, status: 'PENDING_PAYMENT', createdAt: new Date().toISOString() }; // ទាញយក បញ្ជី Pending Orders ចាស់ៗមកបូកបញ្ចូលគ្នា (Multi-Orders Support) let pendingOrdersList = JSON.parse(localStorage.getItem('pendingOrdersList')) || []; pendingOrdersList.push(newOrder); // រក្សាទុកក្នុង LocalStorage localStorage.setItem('pendingOrdersList', JSON.stringify(pendingOrdersList)); localStorage.setItem('pendingOrder', JSON.stringify(newOrder)); // សម្រាប់ Backup // Clear Cart ចាស់ localStorage.removeItem('cartData'); // បញ្ជូនភ្ញៀវទៅទំព័ររង់ចាំទូទាត់ប្រាក់ window.location.href = 'waiting-payment.html'; }
