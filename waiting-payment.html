async function confirmAndSend() { if (!selectedImageFile) { showToast('⚠️ សូមជ្រើសរូបភាពបង្កាន់ដៃទូទាត់ជាមុនសិន!'); return; } if (isSending) { showToast('⏳ កំពុងផ្ញើ... សូមរង់ចាំបន្តិច'); return; } const sendBtn = document.getElementById('sendImageBtn'); sendBtn.disabled = true; sendBtn.classList.add('opacity-70', 'cursor-not-allowed'); sendBtn.innerHTML = ` កំពុងផ្ញើរូប...`; isSending = true; try { // សាងសង់ Caption ពី Order ដោយផ្ទាល់ (ដើម្បីកុំឱ្យខ្វះ Caption) const currentOrder = JSON.parse(localStorage.getItem('pendingOrder')); let finalCaption = 'រូបភាពបង្កាន់ដៃទូទាត់'; if (currentOrder && currentOrder.details && currentOrder.details.customer) { const cust = currentOrder.details.customer; finalCaption = `🛒 ការបញ្ជាទិញពី Mini App\n`; finalCaption += `👤 អតិថិជន: ${cust.name || 'N/A'}\n`; finalCaption += `📞 ទូរស័ព្ទ: ${cust.phone || 'N/A'}\n`; finalCaption += `📍 អាសយដ្ឋាន: ${cust.address || 'N/A'}`; } let formData = new FormData(); formData.append('chat_id', CHAT_ID); formData.append('photo', selectedImageFile); formData.append('caption', finalCaption); // ដាក់ Caption ថ្មីដែលទើបសាងសង់ const response = await fetch(`https://api.telegram.org/bot${BOT_TOKEN}/sendPhoto`, { method: 'POST', body: formData }); const result = await response.json(); if (result.ok) { localStorage.removeItem('pendingOrder'); localStorage.removeItem('pendingOrdersList'); localStorage.removeItem('cartData'); localStorage.removeItem('pendingOrderText'); closeUploadModal(); showToast('✅ ផ្ញើរូបភាពទៅ Telegram Bot ដោយជោគជ័យ!'); setTimeout(() => { document.getElementById('successModal').classList.add('flex'); }, 300); } else { console.error('Telegram Error:', result); showToast(`❌ បរាជ័យ: ${result.description || 'Unknown Error'}`); } } catch (error) { console.error('Network Error:', error); showToast('❌ មានបញ្ហាបណ្តាញ ឬ Server មិនដំណើរការ!'); } sendBtn.innerHTML = `🚀ផ្ញើរូបទៅ Bot`; sendBtn.disabled = false; sendBtn.classList.remove('opacity-70', 'cursor-not-allowed'); isSending = false; }
